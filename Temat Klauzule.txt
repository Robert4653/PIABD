Zad.1.Policz uczniów urodzonych w poszczególnych latach
SELECT year('data_urodzenia') AS rok_urodzenia_ucznia,
COUNT(*) AS liczba_urodzeń_uczniów 
FROM 4bgr1.uczniowie 
GROUP by year('data_urodzenia')
ORDER BY rok_urodzenia_ucznia ASC;
